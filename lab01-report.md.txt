5. Креативні завдання (5 нестандартних запитів)
-- 1. Динамічна сегментація асортименту та розрахунок вартості складських запасів із ПДВ
-- Бізнес-логіка: фінансова оцінка капіталу у складських залишках із розбивкою на сегменти
SELECT 
    product_name AS "Товар",
    units_in_stock AS "Кількість",
    unit_price AS "Ціна без ПДВ",
    ROUND(unit_price * 1.2, 2) AS "Ціна з ПДВ",
    (unit_price * units_in_stock) AS "Капіталізація складу",
    CASE 
        WHEN unit_price < 5000 THEN 'Мас-маркет'
        WHEN unit_price BETWEEN 5000 AND 25000 THEN 'Оптимальний вибір'
        ELSE 'Флагманський сегмент'
    END AS "Ринковий сегмент"
FROM products
WHERE units_in_stock > 0
ORDER BY "Капіталізація складу" DESC
LIMIT 15;

-- 2. Кадрова аналітика співробітників (класифікація стажу та ієрархії управління)
-- Бізнес-логіка: формування HR-профілю працівників
SELECT 
    first_name || ' ' || last_name AS "Співробітник",
    title AS "Посада",
    salary AS "Оклад, грн",
    CASE 
        WHEN reports_to IS NULL THEN 'Топ-менеджмент'
        ELSE 'Лінійний персонал (ID керівника: ' || reports_to || ')'
    END AS "Рівень ієрархії",
    CASE 
        WHEN hire_date < '2022-01-01' THEN 'Ветеран компанії'
        WHEN hire_date < '2024-01-01' THEN 'Досвідчений фахівець'
        ELSE 'Новий співробітник'
    END AS "Категорія стажу"
FROM employees
ORDER BY salary DESC NULLS LAST;

-- 3. Автоматична генерація текстових карток клієнтів для експорту в CRM
-- Бізнес-логіка: безпечне об'єднання полів із обробкою NULL-значень через COALESCE
SELECT 
    customer_id AS "ID",
    contact_name || ' [' || COALESCE(company_name, 'Фізична особа') || '] — м. ' || city || 
    ' | Тел: ' || COALESCE(phone, 'не вказано') || 
    ' | Email: ' || COALESCE(email, 'відсутній') AS "Зведене досьє клієнта"
FROM customers
ORDER BY customer_type DESC, city ASC;

-- 4. Аудит аномалій та потенційних помилок у таблиці товарів
-- Бізнес-логіка: виявлення неліквідних залишків та дефіцитних позицій
SELECT 
    product_id,
    product_name,
    unit_price,
    units_in_stock,
    discontinued,
    CASE 
        WHEN unit_price <= 0 THEN 'Помилка: некоректна ціна'
        WHEN discontinued AND units_in_stock > 20 THEN 'Увага: неліквідний залишок знятого товару'
        WHEN NOT discontinued AND units_in_stock = 0 THEN 'Дефіцит: потрібна закупівля'
        ELSE 'Норма'
    END AS "Статус аудиту"
FROM products
WHERE unit_price <= 0 
   OR (discontinued AND units_in_stock > 20)
   OR (NOT discontinued AND units_in_stock = 0);

-- 5. Індекс пріоритетності обробки замовлень для відділу логістики
-- Бізнес-логіка: визначення черговості відвантаження замовлень
SELECT 
    order_id AS "№ Замовлення",
    order_date AS "Дата створення",
    status AS "Поточний статус",
    CASE 
        WHEN status = 'pending' AND shipped_date IS NULL THEN 'КРИТИЧНИЙ: очікує підтвердження'
        WHEN status = 'processing' AND shipped_date IS NULL THEN 'ВИСОКИЙ: готується до відправки'
        WHEN status = 'shipped' THEN 'СЕРЕДНІЙ: у дорозі'
        ELSE 'ЗАВЕРШЕНО'
    END AS "Пріоритет логістики"
FROM orders
WHERE status != 'cancelled'
ORDER BY order_date ASC
LIMIT 25;
Результат: Усі 5 нестандартних аналітичних запитів успішно протестовані та повернули змістовні управлінські звіти.
